  \IfFileExists{assets/#1}{%

% Card primitives
% #1 x, #2 y-top, #3 width, #4 height, #5 title, #6 body
\newcommand{\CardAt}[6]{%
  \draw[fill=white, draw=Line, line width=0.35mm, rounded corners=2.4mm]
    (#1,#2) rectangle ++(#3,-#4);
  \node[anchor=north west, inner sep=0pt]
    at ($(#1,#2)+(3.2,-3.0)$)
    {\parbox[t]{\dimexpr#3mm-6.4mm\relax}{%
      {\SectionText\color{Navy}#5}\par\vspace{1.0mm}%
      {\PosterText #6}%
    }};
}

% Technology tile
% #1 x, #2 y, #3 asset, #4 label
\newcommand{\TechAt}[4]{%
  \node[anchor=north, align=center, inner sep=0pt] at (#1,#2)
  {%
    \AssetImage{#3}{height=15mm,keepaspectratio}{14mm}\\[0.4mm]
    {\TinyPosterText\color{Muted}#4}%
  };
}
